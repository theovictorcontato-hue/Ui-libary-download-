Full Example

lua · complete 
local PowerUILib = loadstring(game:HttpGet("https://pastefy.app/7mEPB0wl/raw"))()

local Window = PowerUILib:CreateWindow({
    Title = "POWER HUB",
    Width = 270,
})

Window:Toggle({
    Title = "Speed",
    Default = false,
    Callback = function(Value)
        local hum = game.Players.LocalPlayer.Character
            and game.Players.LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        if hum and Value then
            hum.WalkSpeed = 30
        elseif hum then
            hum.WalkSpeed = 16
        end
    end,
})

Window:Toggle({
    Title = "Fly",
    Callback = function(Value)
        print("Fly:", Value)
    end,
})

Window:Slider({
    Title = "WalkSpeed",
    Min = 16,
    Max = 100,
    Default = 16,
    Callback = function(Value)
        local hum = game.Players.LocalPlayer.Character
            and game.Players.LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        if hum then hum.WalkSpeed = Value end
    end,
})

Window:Button({
    Title = "Print OK",
    Callback = function()
        print("PowerUILib ok")
    end,
})

Window:Label({ Title = "PowerUILib · client-side" })
